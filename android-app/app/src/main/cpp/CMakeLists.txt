cmake_minimum_required(VERSION 3.22.1)
project(liangju LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Keep mobile inference optimized in debug APKs too. Unoptimized matmul can time out.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-O3)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp llama)
if(ANDROID)
    add_library(liangju SHARED engine.cpp jni.cpp)
    target_link_libraries(liangju PRIVATE llama android log)
    target_link_options(liangju PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(liangju-smoke engine.cpp smoke.cpp)
    target_link_libraries(liangju-smoke PRIVATE llama)
endif()
